cmake_minimum_required(VERSION 3.8)
project(arrival_fixture)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(pluginlib REQUIRED)
add_executable(arrival_planner planner.cpp)
target_compile_features(arrival_planner PUBLIC cxx_std_17)
ament_target_dependencies(arrival_planner nav2_core nav2_costmap_2d pluginlib)

add_executable(pivot_model pivot_model.cpp)
target_compile_features(pivot_model PUBLIC cxx_std_17)
target_include_directories(pivot_model PRIVATE ../../navigation ../../navigation/p2_nav_controller/include/p2_nav_controller)

add_executable(tracking_replay tracking_replay.cpp)
target_compile_features(tracking_replay PUBLIC cxx_std_17)
target_include_directories(tracking_replay PRIVATE ../../navigation)
ament_target_dependencies(tracking_replay nav2_core nav2_costmap_2d pluginlib)
