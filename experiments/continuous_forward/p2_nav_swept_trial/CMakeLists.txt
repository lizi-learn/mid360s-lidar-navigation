cmake_minimum_required(VERSION 3.8)
project(p2_nav_swept_trial)
find_package(ament_cmake REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(p2_smac_continuous REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
add_library(p2_nav_swept_trial SHARED src/swept_lattice.cpp)
target_compile_features(p2_nav_swept_trial PUBLIC cxx_std_17)
target_include_directories(p2_nav_swept_trial PRIVATE include)
ament_target_dependencies(p2_nav_swept_trial nav2_core pluginlib)
target_include_directories(p2_nav_swept_trial PRIVATE ${nav2_smac_planner_INCLUDE_DIRS})
find_library(P2_SMAC_LATTICE nav2_smac_planner_lattice PATHS /opt/ros/humble/lib REQUIRED)
target_link_libraries(p2_nav_swept_trial ${P2_SMAC_LATTICE})
pluginlib_export_plugin_description_file(nav2_core plugin.xml)
install(TARGETS p2_nav_swept_trial LIBRARY DESTINATION lib)

add_executable(lattice_sweep_regression tests/lattice_sweep_regression.cpp)
target_compile_features(lattice_sweep_regression PUBLIC cxx_std_17)
target_include_directories(lattice_sweep_regression PRIVATE ${nav2_smac_planner_INCLUDE_DIRS})
ament_target_dependencies(lattice_sweep_regression nav2_costmap_2d rclcpp_lifecycle)
target_link_libraries(lattice_sweep_regression ${P2_SMAC_LATTICE} ${CMAKE_DL_LIBS})
install(TARGETS lattice_sweep_regression RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
