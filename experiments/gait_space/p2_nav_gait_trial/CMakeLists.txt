cmake_minimum_required(VERSION 3.8)
project(p2_nav_gait_trial)
find_package(ament_cmake REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(p2_smac_gait REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
add_library(p2_nav_gait_trial SHARED src/swept_lattice.cpp)
target_compile_features(p2_nav_gait_trial PUBLIC cxx_std_17)
target_include_directories(p2_nav_gait_trial PRIVATE include)
ament_target_dependencies(p2_nav_gait_trial nav2_core pluginlib)
target_include_directories(p2_nav_gait_trial PRIVATE ${nav2_smac_planner_INCLUDE_DIRS})
find_library(P2_SMAC_LATTICE nav2_smac_planner_lattice PATHS "${p2_smac_gait_DIR}/../../../lib" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(p2_nav_gait_trial ${P2_SMAC_LATTICE})
pluginlib_export_plugin_description_file(nav2_core plugin.xml)
install(TARGETS p2_nav_gait_trial LIBRARY DESTINATION lib)
foreach(test_name gait_contract_regression gait_hook_regression)
 add_executable(${test_name} tests/${test_name}.cpp)
 target_compile_features(${test_name} PUBLIC cxx_std_17)
 target_include_directories(${test_name} PRIVATE include ${nav2_smac_planner_INCLUDE_DIRS})
 ament_target_dependencies(${test_name} nav2_costmap_2d rclcpp_lifecycle)
 target_link_libraries(${test_name} ${P2_SMAC_LATTICE} ${CMAKE_DL_LIBS})
 install(TARGETS ${test_name} RUNTIME DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
