cmake_minimum_required(VERSION 3.8)
project(p2_smac_gait)
find_package(ament_cmake REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(ompl REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(angles REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/fetch_pinned_smac.py
  --output ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream RESULT_VARIABLE fetch_status)
if(NOT fetch_status EQUAL 0)
  message(FATAL_ERROR "Could not prepare the verified continuous-lattice sources")
endif()
add_library(nav2_smac_planner_lattice SHARED ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/smac_planner_lattice.cpp ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/a_star.cpp ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/smoother.cpp ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/collision_checker.cpp ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/analytic_expansion.cpp ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/node_hybrid.cpp ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/node_lattice.cpp ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/costmap_downsampler.cpp ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/node_2d.cpp ${CMAKE_CURRENT_BINARY_DIR}/verified-upstream/src/node_basic.cpp ${CMAKE_CURRENT_SOURCE_DIR}/src/lattice_gait_constraints.cpp)
target_compile_features(nav2_smac_planner_lattice PUBLIC cxx_std_17)
target_include_directories(nav2_smac_planner_lattice PRIVATE ${nav2_smac_planner_INCLUDE_DIRS} ${Eigen3_INCLUDE_DIRS} ${OMPL_INCLUDE_DIRS})
ament_target_dependencies(nav2_smac_planner_lattice rclcpp rclcpp_action rclcpp_lifecycle std_msgs visualization_msgs nav2_util nav2_msgs nav_msgs geometry_msgs builtin_interfaces tf2_ros nav2_costmap_2d nav2_core pluginlib angles eigen3_cmake_module)
target_link_libraries(nav2_smac_planner_lattice ${OMPL_LIBRARIES})
install(TARGETS nav2_smac_planner_lattice LIBRARY DESTINATION lib)
ament_environment_hooks(env/gait_lattice.sh)
ament_package()
