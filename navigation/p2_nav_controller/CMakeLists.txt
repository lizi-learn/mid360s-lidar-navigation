cmake_minimum_required(VERSION 3.8)
project(p2_nav_controller)
find_package(ament_cmake REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(pluginlib REQUIRED)
add_library(p2_nav_controller SHARED src/settled_goal_rpp.cpp)
target_compile_features(p2_nav_controller PUBLIC cxx_std_17)
target_include_directories(p2_nav_controller PRIVATE include)
ament_target_dependencies(p2_nav_controller nav2_regulated_pure_pursuit_controller nav2_core pluginlib)
pluginlib_export_plugin_description_file(nav2_core p2_nav_controller.xml)
add_library(p2_nav_planner SHARED src/exact_heading_lattice.cpp)
target_compile_features(p2_nav_planner PUBLIC cxx_std_17)
target_include_directories(p2_nav_planner PRIVATE include)
ament_target_dependencies(p2_nav_planner nav2_smac_planner nav2_core pluginlib)
install(TARGETS p2_nav_controller p2_nav_planner LIBRARY DESTINATION lib)
pluginlib_export_plugin_description_file(nav2_core p2_nav_planner.xml)
ament_package()
