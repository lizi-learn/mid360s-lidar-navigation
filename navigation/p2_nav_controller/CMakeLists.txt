cmake_minimum_required(VERSION 3.8)
project(p2_nav_controller)
find_package(ament_cmake REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav2_collision_monitor REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(p2_nav_controller SHARED src/settled_goal_rpp.cpp src/position_goal_checker.cpp src/settled_heading_goal_checker.cpp src/rotation_progress_checker.cpp)
target_compile_features(p2_nav_controller PUBLIC cxx_std_17)
target_link_options(p2_nav_controller PRIVATE "-Wl,--no-undefined")
target_include_directories(p2_nav_controller PRIVATE include)
ament_target_dependencies(p2_nav_controller nav2_regulated_pure_pursuit_controller nav2_core pluginlib)
pluginlib_export_plugin_description_file(nav2_core p2_nav_controller.xml)
pluginlib_export_plugin_description_file(nav2_core p2_nav_goal_checker.xml)
pluginlib_export_plugin_description_file(nav2_core p2_nav_progress_checker.xml)
add_library(p2_nav_planner SHARED src/exact_heading_lattice.cpp src/position_only_lattice.cpp)
target_compile_features(p2_nav_planner PUBLIC cxx_std_17)
target_include_directories(p2_nav_planner PRIVATE include)
# Humble's planner libraries each bundle common Smac statics. Linking the
# generic and lattice libraries together can double-destroy the heuristic cache.
find_library(P2_SMAC_LATTICE NAMES nav2_smac_planner_lattice REQUIRED)
target_include_directories(p2_nav_planner PRIVATE ${nav2_smac_planner_INCLUDE_DIRS})
ament_target_dependencies(p2_nav_planner nav2_core pluginlib)
target_link_libraries(p2_nav_planner ${P2_SMAC_LATTICE})
install(TARGETS p2_nav_controller p2_nav_planner LIBRARY DESTINATION lib)
pluginlib_export_plugin_description_file(nav2_core p2_nav_planner.xml)
add_executable(observed_collision_monitor src/observed_collision_monitor.cpp)
target_compile_features(observed_collision_monitor PUBLIC cxx_std_17)
ament_target_dependencies(observed_collision_monitor nav2_collision_monitor nav2_msgs std_msgs)
install(TARGETS observed_collision_monitor RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
